   >   let dotProduct = 0, norm1 = 0, norm2 = 0
   >   for (let i = 0; i < vec1.length; i++) {
   >     dotProduct += vec1[i] * vec2[i]
   >     norm1 += vec1[i] * vec1[i]

   >   keywordRows.forEach((row, index) => {

   >     // ...累加到合并映射
   >   })

   >   // 2. 处理向量结果（同一结果时累加 RRF 分数）

   >   const executing = new Set()
   >

   >     const promise = asyncFn(item).then(result => {
